\section{Colored Braided Operads}\label{sec:colbraidop}

\textcolor{red}{In this section we provide two original definitions of colored braided operads and develop the theory connecting them.
We start with a first direct definition that allows us to understand colored braided operads in a similar way as colored 
symmetric operad from their classical Definition \ref{coloroperad}.
However, a generalization of colored braided operads to the $\infty$-categorical context requires a categorical definition.
To provide this, we follow an approach similar to the construction of the forgetful functor $\Ox \to \Fin$ from a colored operad $\O$, outlined in Section \ref{sec:operads}.
Doing so, we draw inspiration from Lurie's treatment of $\infty$-operads and his illustrations of the little disks operad as developed in \cite[Section 2]{ha}.
Concretely, we define a category $\Bx$ from a colored braided operad $\B$ and a base category $\Brx$, such that there is a forgetful functor $\pi \colon \Bx \to \Brx$ that encodes all the necessary datum of $\B$.
As a first example of a colored braided operad $\B$ defined via a functor into $\Brx$, we see how an op-fibration in this setting induces a braided monoidal structure on the underlying category $\B$.
Afterwards, we examine the general properties of $\pi$ in more detail and do the work to show that it indeed gives us the categorical definition we are looking for.}

\subsection{Colored Braided Operads}

In this section we give our first definition of a colored braided operad and
consider some examples. We begin with some necessary terminology.

We introduce a slightly more general variant of the space of little disks
$\E_2(n)$ introduced in Example \ref{eg:operads}.

Let $\dd^2$ denote the closed unit disk in $\R^2$. We recall a little disk from
Example \ref{eg:operads} which is defined as an embedding $f \colon \dd^2 \to
\dd^2$ of the following form:
\[
    f(r_1, r_2) \coloneqq (c_1, c_2) + \lambda(r_1, r_2)
\]
for $(c_1, c_n) \in \dd^2$ and $\lambda \in \R$ such that
\[
    0 < \lambda \leq 1 - \sum_{i=1}^{2} c_i^2.     
\]
We denote the set of little disks by $\ld$. Further for a finite set $I$, we
equip the space of continuous maps $C\left(\bigsqcup_{i \in I} \dd^2,
\dd^2\right)$ with the compact-open topology.

\begin{defin}\label{def:ilabeled} For a finite set $I$, we define a topological
space
\[
    \dd(I) \coloneqq \left\{ \bigsqcup_{i \in I} f_i \in C\left(\bigsqcup_{i \in I} \dd^2, \dd^2\right) \colon \inter(f_i) \cap \inter(f_j) = \emptyset  \; \text{and} \; f_i, f_j \in \ld \; \text{ for all } i \not= j\right\}.
\]
with the subspace topology. We refer to the points of $\dd(I)$ as $I$-labeled
configurations. 
% to be the subspace of $C\left(\coprod_{I} D, D\right)$ consisting of those
% embeddings such that the restriction to each component of the coproduct is a
% scale-translate embedding, and the disks are embedded jointly. 
For a map of finite sets $\alpha \colon I \to J$, a $J$-labeled configuration
$\widetilde{J}$ along with $\alpha^{-1}(j)$-labeled configurations 
$$\left\{\widetilde{\alpha^{-1}(j)}\right\}_{j \in J},$$ we obtain an
$I$-labeled configuration $\widetilde{I}$ by nesting the various disks in
analogy to the definition of $E_2$ in Example \ref{eg:operads}. Hence, for any
map of finite sets $\alpha \colon I \to J$ we obtain a composition map:

\begin{equation}\label{eq:defcomp}
    \gamma: D(J) \times \prod_{j \in J} D(\alpha^{-1}(j)) \to D(I).
\end{equation}
\end{defin}

In other words, we simply consider the little disks in $\E_2(n)$ as labeled by $I$
instead of $\{1,...,n\}$. 

\begin{defin}
    A category $\C$ is called a groupoid if every morphism in $\C$ has an
    inverse. We denote the category of groupoids by $\Grp$.
\end{defin}
% \begin{defin} Let $I$ be a finite set. For a bijection of finite sets $\alpha
%     \colon I \cong \langle n \rangle^{\circ}$, we call a configuration $c \in
%     \E_2(n)$ an \emph{$I$-labeled} configuration. \end{defin}

% \begin{defin} Let $I$ be a finite set, $c$ and $c'$ $I$-labeled configurations
%     corresponding to the same bijection $I \cong \langle n \rangle^{\circ}$. A
%     path from $c$ to $c'$ is a path between the configurations in $\E_2(n)$,
%     i.e. a continuous map
%     \[
%         f \colon [0,1] \to \E_2(n)
%     \] with $f(0)=c$ and $f(1)=c'$.
% \end{defin}


\begin{defin}
    We define the $\Grp$-enriched category $\Br$ of {\em braided sets} as follows:
    \begin{enumerate}[(i)]
        \item The objects of $\Br$ are finite sets.
        \item A morphism $\widetilde{\alpha} \colon I \to J$ in $\Br$ consists
        of a map of finite sets  $\alpha \colon I \to J$ and a collection of
        labeled configurations
        \[
            \left\{\widetilde{\alpha^{-1}(j)} \in D(\alpha^{-1}(j))\right\}_{j \in J}
        \]

        % $$\left\{\widetilde{\alpha}_j \in
        % \E_2\left(|\alpha'^{-1}(j)|\right)\right\}_{j \in J}$$
        \begin{enumerate}
            \item For morphisms
        \[
            \widetilde{\alpha} \colon I \to J, \quad \widetilde{\beta} \colon J \to K
        \]
        
        the composition $\widetilde{\beta} \circ \widetilde{\alpha}$ consists of
        the map of finite sets
        \[
            \beta \circ \alpha \colon I \to K
        \]
        % and for all $k \in K$ with
        % \[
        %     \beta^{-1}(k)=\{j_1, \dots, j_p\}
        % \]
        and the $(\beta \circ \alpha)^{-1}(k)$-labeled configuration
        \[
            \widetilde{(\beta \circ \alpha)^{-1}(k)} \coloneqq \gamma\left(\widetilde{\beta^{-1}(k)}, (\widetilde{\alpha^{-1}(j)})_{j \in \beta^{-1}(k)}\right),%}\widetilde{\alpha^{-1}(j_1)}, \dots, \widetilde{\alpha^{-1}(j_p)}\right),
        \]
        where $\gamma$ is the composition map from
        (\ref{eq:defcomp}).
        
        % analogously to the composition map in $\Ex$ as defined above.
        \item The identity morphism $\id_I$ is given by the identity on the
        finite set $I$ and the collection of identity configurations $\{\id^{(i)}
        \in D(\{i\})\}_{i \in I}$.
        \end{enumerate}
        \item For morphisms $\widetilde{\alpha}, \widetilde{\beta} \colon I \to
        J$ in $\Br$ with the same underlying map $\alpha = \beta$, a
        $2$-morphism in $\Br$ consists of a family of homotopy classes of paths in $\dd(\alpha^{-1}(j))$
        \[
            \left\{\eta_j \colon \widetilde{\alpha^{-1}(j)} \Rightarrow \widetilde{\beta^{-1}(j)}\right\}_{j \in J}.
        \]
    \end{enumerate}
\end{defin}
This defines a strict $2$-category where all $2$-morphisms are invertible, i.e. a $\Grp$-enriched category. Note
that a morphism $I \to \{1\}$ in $\Br$ simply consists of an $I$-labeled
configuration $\widetilde{I}$ which we usually use to denote the morphism. 

\begin{nota}
    To avoid clumsy notation, for a morphism $\widetilde{\alpha} \colon I \to J$ in $\Br$, we write
    $$\alpha_j \coloneqq \alpha^{-1}(j)$$ 
    for the preimage and 
    $$\widetilde{\alpha_j} \coloneqq \widetilde{\alpha^{-1}(j)}$$
    for the configurations.
\end{nota}

    

We are now ready to define the concept of a colored braided operad:   

\begin{defin}\label{def:col-br-op} A {\em colored braided operad} $\B$ consists
    of the following data:
    \begin{enumerate}[(i)]
        \item A collection $\{X,Y,Z, \dots\}$ of colors.
        \item For every $I$-labeled configuration $\widetilde{I}$, every
        collection of colors $\{X_i\}_{i \in I}$, and every color $Y$ in $\B$, a
        set
        \[
            \Mul_{\B}^{\widetilde{I}}(\{X_i\}_{i \in I}, Y),
        \]
        called the set of morphisms from $\{X_i\}_{i \in I}$ to $Y$.
        \item For every triangle in $\Br$:



        % $\alpha \colon I \to J$ being a morphism in $\Br$ with its collection
        % of configurations labeled by the preimages of $\alpha$: 

        % $$\widetilde{\alpha} = (\widetilde{I_j})_{j \in J},$$
        
        % where $I_j$ denotes the fiber over $j$ for all $j \in J$, a
        % $2$-morphism $\eta$:
        \begin{center}
            \begin{tikzpicture}
                % Define node positions
                \node (B) at (0, 2) {$I$}; \node (C) at (3, 2) {$J$}; \node (A)
                at (1.5, 0) {$\{ 1 \}$};
                
                % Draw arrows
                \draw[->] (B) -- (C) node[midway, above] {$\widetilde{\alpha}$};
                \draw[->] (B) -- (A) node[midway, left] {$\widetilde{I}$};
                \draw[->] (C) -- (A) node[midway, right] {$\widetilde{J}$};
                
                % Natural transformation arrow
                \node[rotate=-40] at (1.5, 1.2) {$\Downarrow \eta$};
                % \node at (1.7, 1.2) {$\eta$};
            \end{tikzpicture}
        \end{center}
        along with collections of colors $\{X_i\}_{i \in I}$, $\{Y_j\}_{j \in
        J}$ and color $Z$, a composition map:
        \[
            \gamma \colon \prod_{j \in J} \Mul_{\B}^{\widetilde{\alpha_j}}\left(\{X_i\}_{i \in \alpha_j}, Y_j\right) \times \Mul_{\B}^{\widetilde{J}}\left(\{Y_j\}_{j\in J}, Z\right) \to \Mul_{\B}^{\widetilde{I}}\left(\{X_i\}_{i \in I}, Z\right)  
        \]
        \item For any $I$-labeled configuration $\widetilde{I}$ with $|I|=1$ and
        color $X$, a morphism

        $$\id_X \in \Mul_{\B}^{\widetilde{I}}(\{X\}, X),$$

        called identity morphism on $X$.
        \end{enumerate}
        The data needs to fulfill the following properties:
        \begin{enumerate}
            \item The identity morphisms are a left and right unit with respect
        to the composition map $\gamma$, i.e. the following diagrams commute:
        \begin{center}
            \begin{tikzcd}
                {\Mul_{\B}^{\widetilde{I}}(\{X_i\}_{i \in I}, Y)} \arrow[r,
                "\cong"] \arrow[d, "\id"] &
                {\Mul_{\B}^{\widetilde{I}}(\{X_i\}_{i \in I}, Y) \times
                \{\id_Y\}} \arrow[d, hook]           \\
                {\Mul_{\B}^{\widetilde{I}}(\{X_i\}_{i \in I}, Y)} &
                {\Mul_{\B}^{\widetilde{I}}(\{X_i\}_{i \in I}, Y) \times
                \Mul_{\B}^{c}((Y), Y)} \arrow[l, "\gamma"]
            \end{tikzcd}
        \end{center}
       
        \begin{center}
            \begin{tikzcd}
                {\Mul_{\B}^{\widetilde{I}}(X_I, Y)} \arrow[r, "\cong"] \arrow[d,
                "\id"] & {\left(\prod_{i \in I} \{\id_{X_i}\}\right) \times
                \Mul_{\B}^{\widetilde{I}}(X_I, Y)} \arrow[d, hook]           \\
                {\Mul_{\B}^{\widetilde{I}}(X_I, Y)} & {\left(\prod_{i \in I}
                \Mul_{\B}^{c}((X_i), X_i)\right) \times
                \Mul_{\B}^{\widetilde{I}}(X_I, Y)} \arrow[l, "\gamma"]
            \end{tikzcd}                                             
        \end{center}

        
        
    
        \item For every sequence of morphisms $I
        \overset{\widetilde{\alpha}}{\to} J \overset{\widetilde{\beta}}{\to} K$
        in $\Br$, $I$-labeled configuration $\widetilde{I}$, $J$-labeled
        configuration $\widetilde{J}$, 2-morphisms $\eta$, $\epsilon$ in $\Br$
        as in the following diagram:

        \begin{center}
            \begin{tikzpicture}
                % Define node positions
                \node (B) at (0, 2) {$I$}; \node (C) at (3, 2) {$J$}; \node (A)
                at (3, -1) {$\{1 \}$}; \node (D) at (6, 2) {$K$};

                % Draw arrows
                \draw[->] (B) -- (C) node[midway, above] {$\widetilde{\alpha}$};
                \draw[->] (C) -- (D) node[midway, above] {$\widetilde{\beta}$};
                \draw[->] (B) -- (A) node[midway, left] {$\widetilde{I}$};
                \draw[->] (C) -- (A) node[midway, right] {$\widetilde{J}$};
                \draw[->] (D) -- (A) node[midway, right] {$\widetilde{K}$};
                % Natural transformation arrow
                \node at (2.0, 1.1) {$\overset{\Leftarrow}{\eta}$}; \node at
                (4.2, 1.1) {$\overset{\Leftarrow}{\epsilon}$};
            \end{tikzpicture}
        \end{center}
        % \begin{center} \begin{tikzcd} {\prod_{j \in J} \Mul_{\O}(X_I, Y_j)
        %     \times \prod_{k \in K} \Mul_{\O}(Y_J, W_k) \times \Mul_{\O}(W_K,
        %     Z)} \arrow[rd, "\id \times \gamma"] \arrow[dd, "(\gamma \circ s)
        %     \times \id"] & \\
        %                     & {\prod_{j \in J} \Mul_{\O}(X_{I_j}, Y_j) \times
        %                     \Mul_{\O}(Y_J, Z)} \arrow[dd] \\
        %         {\prod_{k \in K} \Mul_{\O}(X_{I_k}, W_k) \times \Mul_{\O}(W_K,
        %         Z)} \arrow[rd] & \\
        %                     & {\Mul_{\O}(X_I, Z)}         
        %     \end{tikzcd} \end{center}

        along with collections of colors $X_I = \{X_i\}_{i \in I}$, $Y_J =
        \{Y_j\}_{j \in J}$, $W_K = \{W_k\}_{k \in K}$ and a color $Z \in \B$,
        the following diagram commutes:
        \begin{center}
        \begin{adjustbox}{max width=\textwidth}
            \begin{tikzcd}
                {\prod_{j \in J} \Mul_{\B}^{\widetilde{\alpha_j}}(X_{I_j}, Y_j)
                \times \prod_{k \in K} \Mul_{\B}^{\widetilde{\beta_k}}(Y_{J_k},
                W_k) \times \Mul_{\B}^{\widetilde{K}}(W_K, Z)} \arrow[rd, "\id
                \times \gamma"] \arrow[dd, "(\gamma \circ s) \times \id"] & \\
                            & {\prod_{j \in J}
                            \Mul_{\B}^{\widetilde{\alpha_j}}(X_{I_j}, Y_j)
                            \times \Mul_{\B}^{\widetilde{J}}(Y_J, Z)} \arrow[dd]
                            \\
                {\prod_{k \in K} \Mul_{\B}^{\widetilde{(\alpha \circ
                \beta)_k}}(X_{I_k}, W_k) \times \Mul_{\B}^{\widetilde{K}}(W_K,
                Z)} \arrow[rd] & \\
                            & {\Mul_{\B}^{\widetilde{I}}(X_I, Z)}         
            \end{tikzcd}
            \end{adjustbox}
        \end{center}
        \end{enumerate}

\end{defin}

\begin{rem}
    Analogously to other variants of an operad, the Definition
    \ref{def:col-br-op} can not only be defined with values in $\Set$ but rather
    be generalized to any symmetric monoidal category $(\C, \otimes)$ by
    replacing the cartesian product with the tensor product in $\C$.
\end{rem}

\begin{eg}\label{eg:trivial-col-br-op} As a first example consider the
    unicolored braided operad $\B$ with color $X$, trivial sets of morphisms for
    all $I$-labeled configurations $\widetilde{I}$: 
    \[
        \Mul_{\B}^{\widetilde{I}}\left(\{X\}_{i \in I}, X\right) = \{ \ast \}
    \]
    and trivial composition map. Clearly this satisfies all properties of a
    colored braided operad and corresponds to the trivial braided operad we
    encountered in Example \ref{eg:braidedoperads}.
\end{eg}


\begin{eg}
    Let $\Bc$ be a colored braided operad with one color $X$. Similarly to the
    symmetric case, $\Bc$ induces a braided operad $\B$ as in Definition
    \ref{def-br-operad}. For $n \in \N$, we set
    \[
        \B(n) = \Mul_{\Bc}^{b_n}(\{X\}_{1 \leq i \leq n}, X),
    \]

    \[
        \theta \colon \B(k) \times \B(j_1) \times \cdots \times \B(j_k) \to \B(j),

    \]
    corresponding to the unique $\E_1$-translation from $b_j$ to $b_k(b_{j_1},
    \dots, b_{j_k})$ as defined in Remark \ref{rem:e1-translation}:

    There is a unit
    \[
        \eta \colon \ast \to \B(1),

    which is defined by the identity morphism $\id_X \in \Mul_{\Bc}^{b_1}(\{X\},
    X)$. The unitality condition is satisfied as the identity morphism $\id_X$
    is the left and right unit for any composition map $\gamma$, so in
    particular $\theta$. The associativity condition is also satisfied as it
    holds for $\Bc$:

    \[

    % \sum_{0 \leq l \leq s} j_l$ and $h_s = \sum_{1 \leq q \leq j_s}

\subsection{Braided Operadic Fibrations}

Colored symmetric operads can be reformulated as categories that are fibered in
a suitably weakened sense over the category $\Fin$ of finite pointed sets. In
order to develop an analogous perspective for colored braided operads, we now
introduce a pointed variant of the category $\Br$ of braided sets from above. 

\begin{defin}
    Let $\Brr$ be the following $\Grp$-enriched category:
    \begin{enumerate}[(i)]
        \item The objects of $\Brr$ are finite sets with a distinguished point
        $\ast$.
        \item A morphism $\widetilde{\alpha}_{\ast} \colon I \to J$ consists of
        a map of finite pointed sets $\alpha \colon I \to J$ with
        $\alpha(\ast)=\ast$ together with a morphism
        \[
            \widetilde{\alpha} \colon \alpha^{-1}(J \setminus \{\ast\}) \to J \setminus \{\ast\}
        \]
        in $\Br$.
        \item For morphisms $\widetilde{\alpha}_{\ast}, \widetilde{\beta}_{\ast}
        \colon I \to J$ with the same underlying map $\alpha = \beta$, a
        $2$-morphism 
        $$\eta_{\ast} \colon \widetilde{\alpha}_{\ast} \to
        \widetilde{\beta}_{\ast}$$ is a $2$-morphism 
        $$\eta \colon \widetilde{\alpha} \Rightarrow \widetilde{\beta}$$ in
        $\Br$.
    \end{enumerate}
    We call $\Brr$ the category of braided pointed sets.
\end{defin}

\begin{defin}
    Let the category of standard braided pointed sets $\Brx$ be the full
    subcategory of $\Brr$ on the objects $\langle n \rangle$, $n \in \N$.
\end{defin}

\begin{rem}
    Note that that $\Brr$ is equivalent to $\Brx$ in the analogous way as $\Fin$
    is equivalent to the category of finite sets with a distinguished point.
\end{rem}

\begin{defin}
    A morphism $\alpha \colon \langle n \rangle \to \langle m \rangle$ in $\Brx$
    is called \emph{active} if it is active in $\Fin$, i.e. $\alpha^{-1}(\ast) =
    \{\ast\}$.
\end{defin}

Analogously to Lurie in \cite{ha} defining a category $\Ox$ from a colored
symmetric operad $\O$ that gives rise to the functorial definition of a colored
operad, we now want to define a category $\Bx$ from a colored braided operad
$\B$:

\begin{defin}\label{def:B-tensor} Let $\B$ be a colored braided operad. We
    define a $\Grp$-enriched category $\Bx$ in the following way:
    \begin{enumerate}[(i)]
        \item The objects of $\Bx$ are finite sequences $\{X_i\}_{1 \leq i \leq
        n}$ of colors of $\B$.
        \item For objects $\{X_i\}_{1 \leq i \leq m}$, $\{Y_j\}_{1 \leq j \leq
        n}$ in $\Bx$, a morphism 
        \[
            \phi \colon \{X_i\}_{1 \leq i \leq m} \to \{Y_j\}_{1 \leq j \leq n}
        \]
        is given by a map $\widetilde{\alpha} \colon \langle m \rangle \to
        \langle n \rangle$ in $\Brx$ together with a family of morphisms
        \[
            \{ \phi_j \in \Mul_{\B}^{\widetilde{\alpha_j}}(\{X_i\}_{i \in \alpha^{-1}(j)}, Y_j)\}_{1 \leq j \leq n}
        \]
        in $\B$. We call $\widetilde{\alpha}$ the underlying morphism of $\phi$.
        \item Composition of morphisms is determined by the composition in
        $\Brx$ and $\B$, i.e. for morphisms $\phi \colon X \to Y$, $\phi' \colon
        Y \to Z$ in $\Bx$  with $\widetilde{\alpha} \colon \langle n \rangle \to
        \langle m \rangle$, $\widetilde{\alpha'} \colon \langle m \rangle \to
        \langle k \rangle$ respectively, we define $\widetilde{\alpha''} =
        \widetilde{\alpha'} \circ \widetilde{\alpha}$ and the family of
        morphisms
        \[
            \{(\phi' \circ \phi)_l \in  \Mul_{\B}^{\widetilde{\alpha''_l}}(\{X_i\}_{i \in \alpha''^{-1}(l)}, Z_l)\}_{1 \leq l \leq k},
        \]
        where $(\phi' \circ \phi)_l$ is applying to $((\phi_i)_{i \in
        \alpha''^{-1}(l)}, \phi'_l)$ the following composition map 
        \[
            \gamma \colon \prod_{j \in \alpha'^{-1}(l)} \Mul_{\B}^{\widetilde{\alpha_j}}(\{X_i\}_{i \in \alpha^{-1}(j)},Y_j) \times \Mul_{\B}^{\widetilde{\alpha'_l}}(\{Y_i\}_{i \in \alpha'^{-1}(l)},Z_l) \to \Mul_{\B}^{\widetilde{\alpha''}_l}(\{X_i\}_{i \in \alpha''^{-1}(l)},Z_l).
        \] 
        \item For morphisms $\phi, \psi \colon \{X_i\}_{1 \leq i \leq n} \to
        \{Y_j\}_{1 \leq j \leq m}$ with underlying morphism
        $\widetilde{\alpha_{\phi}}$ and $\widetilde{\alpha_{\psi}}$
        respectively, a $2$-morphism 

        $$\bar{\eta} \colon \phi \to \psi$$

        consist of
        a $2$-morphism 
        $$\widetilde{\eta} \colon \widetilde{\alpha_{\phi}} \Rightarrow \widetilde{\alpha_{\psi}}$$

        in $\Brx$ such that for all $1 \leq j \leq m$ the composition maps $\gamma$ corresponding to the
        following diagram:
        \begin{center}
            \begin{tikzpicture}
                % Define node positions
                \node (B) at (0, 2) {$\alpha_{\phi_j}$}; \node (C) at (3, 2)
                {$\alpha_{\psi_j}$}; \node (A) at (1.5, 0) {$\{j\}$};
                % Draw arrows
                \draw[->] (B) -- (C) node[midway, above] {$\id$}; \draw[->] (B)
                -- (A) node[midway, left] {$\widetilde{\alpha_{\phi_j}}$};
                \draw[->] (C) -- (A) node[midway, right]
                {$\widetilde{\alpha_{\psi_j}}$};
                % Natural transformation arrow
                \node[rotate=-40] at (1.4, 1.2) {$\Downarrow \widetilde{\eta}_j$};
                % \node at (1.7, 1.2) {$\eta$};
            \end{tikzpicture}
        \end{center}
        satisfies $\gamma((\id_{X_i})_{i \in \alpha_{\phi_j}}, \psi_j) =
        \phi_j$. The composition of $2$-morphisms is well-defined since for
        $2$-morphisms $\bar{\eta} \colon \phi \Rightarrow \psi$, $\bar{\epsilon} \colon \psi \Rightarrow
        \chi$, as well as composition maps $\gamma, \gamma'$ corresponding to
        $\widetilde{\eta}_j$ and $\widetilde{\epsilon}_j$, the following holds for all $1 \leq j \leq
        m$:
        \begin{align*}
            \phi_j &= \gamma ((\id_{X_i})_{i \in \alpha_{\psi_j}}, \psi_j) \\
            &= \gamma((\id_{X_i})_{i \in \alpha_{\psi_j}}, \gamma'((\id_{X_i})_{i \in \alpha_{\chi_j}}, \chi_j)) \\
            &= ((\id_{X_i})_{i \in \alpha_{\chi_j}}, \chi_j).
        \end{align*}

    \end{enumerate}
\end{defin}

\begin{defin}
Let $\B$ be a colored braided operad, then we define a forgetful functor
\[
    \pi \colon \Bx \to \Brx,
\]
as follows:
\begin{align*}
    [X_1, \dots, X_n] &\mapsto \langle n \rangle \\
    (f \colon [X_1, \dots, X_n] \to [X_1', \dots, X_m']) &\mapsto (\alpha_f \colon \langle n \rangle \to \langle m \rangle) \\
    (\eta \colon \psi \Rightarrow \phi) &\mapsto (\widetilde{\eta} \colon \alpha_{\phi} \Rightarrow \alpha_{\psi})
\end{align*}
where $\alpha_f$ refers to the morphism in $\Brx$ underlying $f$ and $\widetilde{\eta}$ the $2$-morphism underlying $\eta$. 
\end{defin}
It is straight
forward to see that this defines a functor.

\begin{eg}\label{eg:br-fib-trivial}
    The category corresponding, under the construction of Definition \ref{def:B-tensor}, 
    to the trivial unicolored braided
    operad from Example \ref{eg:trivial-col-br-op} is the category $\Brx$.
    The defined forgetful functor is the identity functor on $\Brx$.
    Further, we recall that there is a forgetful functor from $\Brx$ to $\Fin$
    satisfying the conditions of a colored symmetric operad which is equivalent to $\E_2$.
    Thus, we note every colored braided operad gives rise to a colored
    symmetric operad.
\end{eg}

Before we characterize the properties of the functor $\pi$, we analyze the
special case when $\pi$ is an op-fibration that satisfies the Segal condition.
In the analogous context of symmetric operads, this conditions singles out
symmetric monoidal categories so that, for braided operads, we would expect to
obtain braided monoidal categories.
