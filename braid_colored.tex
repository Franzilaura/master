\begin{defin}
    A morphism $\alpha \colon \langle n \rangle \to \langle m \rangle$ in $\Ex$
    is called active if it is active in $\Fin$, i.e. $\alpha^{-1}(\ast) =
    \{\ast\}$.
\end{defin}

\begin{defin}
    Let $I$ be a finite set. We call a configuration $c \in \E_2(n)$ $I$-labeled
    according to $\alpha$ if there is a bijective map $\alpha \colon I \cong
    \langle n \rangle^{\circ}$. In most cases $\alpha$ is not important to us,
    so we just say $c$ is $I$-labeled.
\end{defin}

    the symmetric case, $\B_{col}$ induces a braided operad $\B$ as in
    Definition \ref{def-br-operad}. For $n \in \N$, we set
    \[
        \B(n) = \Mul_{\B_{col}}^{b_n}(\{X\}_{1 \leq i \leq n}, X)
    \]
    where $b_n$ is our basepoint configuration in $\E_2(n)$ as chosen in
    Construction \ref{con:unicover}. Further, we define the composition map
    \[
        \theta \colon \B(k) \times \B(j_1) \times \cdots \times \B(j_k) \to \B(j)
    \]
    where $j = j_1 + \cdots + j_k$, from the composition map $\gamma$ in
    $\B_{col}$:
    \[
        \gamma \colon  \prod_{1 \leq i \leq k} \Mul_{\B_{col}}^{b_{j_i}}\left(\{X\}^{j_i}, X\right) \times \Mul_{\B_{col}}^{b_k}\left(\{X\}^k, X\right) \to \Mul_{\B_{col}}^{b_j}\left(\{X\}^j, X\right)
    \]
    corresponding to the unique $\E_1$-translation from $b_j$ to $b_k(b_{j_1},
    \dots, b_{j_k})$ as defined in Remark \ref{rem:e1-translation}:

    \[
