\section{Colored Braided Operads}

In this section, we will define colored braided operads and will aim to find a op-fibration
that encodes all the necesaary data to give an alternative definition.
First, let us recall the definition of $\Ex$ and build a related category $\Br$ 
that will serve as the base category for our op-fibration later on.

\begin{rem}
    For each $n \in \N$ we have an object $\langle n \rangle$. 
    A mophism $\langle n \rangle \to \langle m \rangle$ in $\Ex$ consists of a morphism 
    $\alpha \colon \langle n \rangle \to \langle m \rangle$ in $\Fin$ and a collection of configurations
    $$\{c_j \in \E_2(|\alpha^{-1}(j)|)\}_{1 \leq j \leq m}$$
    where each configuration is labeled by $\alpha^{-1}(j))$.
    So for example, 
    \begin{align*}
        \alpha \colon \langle 3 \rangle &\to \langle 2 \rangle \\ 1 &\mapsto 1 \\
        2 &\mapsto 2 \\
        3 &\mapsto 1 \\
    \end{align*}
    together with the configurations:
    \begin{center}
    \begin{tikzpicture}
                        % First element (left)
                        \draw[thick] (-3,0) circle (2);
                        \draw[thick] (-3.8, 1) circle (0.5);
                        \draw[thick] (-3, -0.5) circle (1);
                        \node at (-3.8, 1) {\small $1$};
                        \node at (-3, -0.5) {\small $3$};
                        \node at (0,-2) {,};
                        % Second element (right)
                        \draw[thick] (3,0) circle (2);
                        \draw[thick] (2,0.7) circle (0.5);
                        \node at (2,0.7) {\small $2$};
        \end{tikzpicture}
    \end{center}
    We denote the trivial configuration with label $x$ by $\id^x \in \E_2(1)$.
    Further, recall the composition in $\Ex$. Let 
    \begin{center}
        $\alpha \colon \langle n \rangle \to \langle m \rangle$ 
        with configurations $\{c_j\}_{1 \leq j \leq m}$ 
    \end{center}
    and 
    \begin{center}
        $\alpha' \colon \langle m \rangle \to \langle l \rangle$ 
        with configurations $\{c'_k\}_{1 \leq k \leq l}$. 
    \end{center}
    Then, the configuration $c'_k$ is labeled 
    by $\alpha'^{-1}=\{j_1^{(k)}, \dots, j_p^{(k)}\}$. So, we obtain a configuration 
    \[
        \gamma(c'_k, c_{j_1}, \dots, c_{j_p})
    \]
    labeled by $(\alpha' \circ \alpha)^{-1}(k)$ for all $1 \leq k \leq l$ 
    giving us the well-defined composition $\alpha' \circ \alpha$
\end{rem}

Next, we partially enrich $\Ex$ with the groupoid structure.

\begin{defin}
    Let $\Br$ be the following partially $\Grp$-enriched category:
    \begin{enumerate}[(i)]
        \item The objects of $\Br$ are finite sets $\langle n \rangle$ for $n \in \N$.
        \item A morphism $\alpha \colon \langle n \rangle \to \langle m \rangle$ in $\Br$ 
        consists of a morphism $\alpha \colon \langle n \rangle \to \langle m \rangle$ 
        in $\Fin$ and a collection of configurations

        $$\{\widetilde{\alpha}_j \in \E_2(|\alpha^{-1}(j)|)\}_{1 \leq j \leq m}$$

        where each configuration $\widetilde{\alpha}_j$ is labeled by $\alpha^{-1}(j)$.
        The composition is defined analogously to the composition map in $\Ex$ as defined above.
        Further, the identity morphism $\id_{\langle n \rangle}$ is given by the collection of 
        trivial configurations $\{\id^j \in \E_2(1)\}_{1 \leq j \leq n}$.
        \item The active morphisms into $\langle 1 \rangle$
        $$\acthom( \langle n \rangle, \langle 1 \rangle)$$
        form the groupoid of homotopy classes of paths in the configuration space $\E_2(n)$.
        The remaining $\hom$-categories are discrete groupoids.
        \todo{correction: the morphisms are a groupoid, not a set. But its discrete if not active to 1}
    \end{enumerate}
\end{defin}

It is clear that this defines a category.
We can extend the definition of $\Br$ to include any finite set $I$.
Note, that that category would be equivalent to $\Br$ in the analogous
way as a $\Fin$ is equivalent to the category of finite sets with a distinguished point.
We are now set up to understand the following definition:

\begin{defin}
    Let $\B$ be a colored operad. We define a category $\Bx$ in the following way:
    \begin{enumerate}[(i)]
        \item The objects of $\Bx$ are finite sequences of colors of $\B$.
        \item Let $\{X_i\}_{1 \leq i \leq m}$ and $\{Y_j\}_{1 \leq j \leq n}$ be objects in $\Bx$. A morphism 
        \[
            \phi \colon \{X_i\}_{1 \leq i \leq m} \to \{Y_j\}_{1 \leq j \leq n}
        \]
        is given by a map $\alpha \colon \langle m \rangle \to \langle n \rangle$ in $\Br$ together with a 
        family of morphisms
        \[
            \{ \phi_j \in \Mul_{\B}^{\widetilde{\alpha_j}}(\{X_i\}_{i \in \alpha^{-1}(j)}, Y_j)\}_{1 \leq j \leq n}
        \]
        in $\B$.
        \item Composition of morphisms is determined by the composition in $\Br$ and $\B$, i.e. for morphisms 
        $\phi \colon X \to Y$ and $\phi' \colon Y \to Z$ in $\Bx$  with $\alpha' \colon \langle n \rangle \to \langle m \rangle$ 
        and $\alpha'' \colon \langle m \rangle \to \langle k \rangle$, we have $\alpha = \alpha''\circ \alpha'$ and
        \[
            \{(\phi' \circ \phi)_l \in  \Mul_{\B}^{\widetilde{\alpha_l}}(\{X_i\}_{i \in \alpha^{-1}(l)}, Z_l)\}_{1 \leq l \leq k}
        \]
        where $(\phi' \circ \phi)_l$ is applying the composition map 
        \[
            \gamma \colon \prod_{j \in \alpha''^{-1}(l)} \Mul_{\B}^{\widetilde{\alpha'_j}}(\{X_i\}_{i \in \alpha'^{-1}(j)},Y_j) \times \Mul_{\B}^{\widetilde{\alpha''_l}}(\{Y_i\}_{i \in \alpha''^{-1}(l)},Z_l) \to \Mul_{\B}^{\widetilde{\alpha}_l}(\{X_i\}_{i \in \alpha^{-1}(l)},Z_l).
        \] 
        %to $((\phi_i)_{i \in \alpha^{-1}(l)}, \phi'_l)$.
    \end{enumerate}
\end{defin}
