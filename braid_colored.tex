We are now set up to understand the following definition:   

\begin{defin}
    A {\em colored braided operad} $\B$ consists of the following data:
    \begin{enumerate}[(i)]
        \item A collection $\{X,Y,Z, \dots\}$ of colors.
        \item For every finite set $I$  and $\widetilde{I}$ an $I$-labeled configuration in $\E_2$ and
        for every collection of colors $\{X_i\}_{i \in I}$ and a color $Y$ in $\B$, there is 
        a set
        \[
            \Mul_{\B}^{\widetilde{I}}(\{X_i\}_{i \in I}, Y)
        \]
        called the set of morphisms from $\{X_i\}_{i \in I}$ to $Y$.
        \item Let $\alpha \colon I \to J$ be finite map together with a collection of 
        configurations labeled by the preimages of $\alpha$, i.e. 
        $$\widetilde{\alpha} = (\widetilde{I_j})_{j \in J}$$
        where $I_j$ denotes the fiber over $j$ for all $j \in J$
        and a path of configurations $\eta$:
        \begin{center}
            \begin{tikzpicture}
                % Define node positions
                \node (B) at (0, 2) {$I$};
                \node (C) at (3, 2) {$J$};
                \node (A) at (1.5, 0) {$\langle 1 \rangle$};
                
                % Draw arrows
                \draw[->] (B) -- (C) node[midway, above] {$\widetilde{\alpha}$};
                \draw[->] (B) -- (A) node[midway, left] {$\widetilde{I}$};
                \draw[->] (C) -- (A) node[midway, right] {$\widetilde{J}$};
                
                % Natural transformation arrow
                \node[rotate=-40] at (1.5, 1.2) {$\Downarrow \eta$};
                % \node at (1.7, 1.2) {$\eta$};
            \end{tikzpicture}
        \end{center}
        Further, let there be
        collections of colors $\{X_i\}_{i \in I}$ and $\{Y_j\}_{j \in J}$ and color $Z$. Then, 
        there is a composition map:
        \[
            \prod_{j \in J} \Mul_{\B}^{\widetilde{I_j}}(\{X_i\}_{i \in I_j}, Y_j) \times \Mul_{\B}^{\widetilde{J}}(\{Y_j\}_{j\in J}, Z) \to \Mul_{\B}^{\widetilde{I}}(\{X_i\}_{i \in I}, Z)  
        \]
        \item There is a collection of morphisms $\{ \id_X \in \Mul_{\B}^{\id^1}(\{X\}, X)\}_{X \in \B}$ being a left 
        and right unit with respect to the composition, i.e. the following diagrams commute:
        \begin{center}
            \begin{tikzcd}
                {\Mul_{\B}^{\widetilde{I}}(\{X_i\}_{i \in I}, Y)} \arrow[r, "\cong"] \arrow[d, "\id"] & {\Mul_{\B}^{\widetilde{I}}(\{X_i\}_{i \in I}, Y) \times \{\id_Y\}} \arrow[d, hook]           \\
                {\Mul_{\B}^{\widetilde{I}}(\{X_i\}_{i \in I}, Y)}                                     & {\Mul_{\B}^{\widetilde{I}}(\{X_i\}_{i \in I}, Y) \times \Mul_{\B}^{\id^1}((Y), Y)} \arrow[l, "\gamma"]
            \end{tikzcd}
        \end{center}
       
        \begin{center}
            \begin{tikzcd}
                {\Mul_{\B}^{\widetilde{I}}(X_I, Y)} \arrow[r, "\cong"] \arrow[d, "\id"] & {(\prod_{i \in I} \{\id_{X_i}\}) \times \Mul_{\B}^{\widetilde{I}}(X_I, Y)} \arrow[d, hook]           \\
                {\Mul_{\B}^{\widetilde{I}}(X_I, Y)}                                     & {(\prod_{i \in I} \Mul_{\B}^{\id^{i}}((X_i), X_i)) \times \Mul_{\B}^{\widetilde{I}}(X_I, Y)} \arrow[l, "\gamma"]
            \end{tikzcd}                                             
        \end{center}
        \item For every sequence of finite maps $I \overset{\alpha}{\to} J \overset{\beta}{\to} K$, 
        with paths $\eta$ and $\epsilon$ as in the following diagram:

        \begin{center}
            \begin{tikzpicture}
                % Define node positions
                \node (B) at (0, 2) {$I$};
                \node (C) at (3, 2) {$J$};
                \node (A) at (3, -1) {$\langle 1 \rangle$};
                \node (D) at (6, 2) {$K$};

                % Draw arrows
                \draw[->] (B) -- (C) node[midway, above] {$\widetilde{\alpha}$};
                \draw[->] (C) -- (D) node[midway, above] {$\widetilde{\beta}$};
                \draw[->] (B) -- (A) node[midway, left] {$\widetilde{I}$};
                \draw[->] (C) -- (A) node[midway, right] {$\widetilde{J}$};
                \draw[->] (D) -- (A) node[midway, right] {$\widetilde{K}$};
                % Natural transformation arrow
                \node at (2.0, 1.1) {$\overset{\Leftarrow}{\eta}$};
                \node at (4.2, 1.1) {$\overset{\Leftarrow}{\epsilon}$};
            \end{tikzpicture}
        \end{center}
        % \begin{center}
        %     \begin{tikzcd}
        %         {\prod_{j \in J} \Mul_{\O}(X_I, Y_j) \times \prod_{k \in K} \Mul_{\O}(Y_J, W_k) \times \Mul_{\O}(W_K, Z)} \arrow[rd, "\id \times \gamma"] \arrow[dd, "(\gamma \circ s) \times \id"] & \\
        %                     & {\prod_{j \in J} \Mul_{\O}(X_{I_j}, Y_j) \times \Mul_{\O}(Y_J, Z)} \arrow[dd] \\
        %         {\prod_{k \in K} \Mul_{\O}(X_{I_k}, W_k) \times \Mul_{\O}(W_K, Z)} \arrow[rd] & \\
        %                     & {\Mul_{\O}(X_I, Z)}         
        %     \end{tikzcd}
        % \end{center}

        and collections of colors
        $X_I = \{X_i\}_{i \in I}$, $Y_J = \{Y_j\}_{j \in J}$, $W_K = \{W_k\}_{k \in K}$ and a color $Z \in \B$ 
        the following diagram commutes:
         \todo{make diagram prettier}
        \begin{center}
        \begin{adjustbox}{max width=\textwidth}
            \begin{tikzcd}
                {\prod_{j \in J} \Mul_{\B}^{\widetilde{I_j}}(X_{I_j}, Y_j) \times \prod_{k \in K} \Mul_{\B}^{\widetilde{J}}(Y_J, W_k) \times \Mul_{\B}^{\widetilde{K}}(W_K, Z)} \arrow[rd, "\id \times \gamma"] \arrow[dd, "(\gamma \circ s) \times \id"] & \\
                            & {\prod_{j \in J} \Mul_{\B}^{\widetilde{I_j}}(X_{I_j}, Y_j) \times \Mul_{\B}^{\widetilde{J}}(Y_J, Z)} \arrow[dd] \\
                {\prod_{k \in K} \Mul_{\B}^{\widetilde{I_k}}(X_{I_k}, W_k) \times \Mul_{\B}^{\widetilde{K}}(W_K, Z)} \arrow[rd] & \\
                            & {\Mul_{\B}(X_I, Z)}         
            \end{tikzcd}
            \end{adjustbox}
        \end{center}
    \end{enumerate}
\end{defin}

\begin{eg}
    Let $\B_{col}$ be a colored braided operad with one color $X$. Similar to the symmetric case,
    $\B_{col}$ induces a braided operad $\B$ as in definition \ref{def-br-operad}.
    For $n \in \N$, we set
    \[
        \B(n) = \Mul_{\B_{col}}^{b_n}(\{X\}_{1 \leq i \leq n}, X)
    \]
    where $b_n$ is our chosen basepoint configuration in $\E_2(n)$ \ref{reference where we define basepoints}.
    Further, we define the composition map
    \[
        \theta \colon \B(k) \times \B(j_1) \times \cdots \times \B(j_k) \to \B(j)
    \]
    where $j = j_1 + \cdots + j_k$, from the composition map $\gamma$ in $\B_{col}$:
    \[
        \gamma\colon  \prod_{1 \leq i \leq k} \Mul_{\B_{col}}^{b_{j_i}}(\{X_i\}^{j_i}, X) \times \Mul_{\B_{col}}^{b_k}(\{X\}^k, X) \overset{\gamma}{\to} \Mul_{\B_{col}}^{b_j}(\{X\}^j, X)
    \]
    corresponding to the following path of configurations:
    \begin{center}
            \begin{tikzpicture}
                % Define node positions
                \node (B) at (0, 2) {$\langle j \rangle$};
                \node (C) at (3, 2) {$\langle k \rangle$};
                \node (A) at (1.5, 0) {$\langle 1 \rangle$};
                
                % Draw arrows
                \draw[->] (B) -- (C) node[midway, above] {$(b_{j_1}, \dots, b_{j_k})$};
                \draw[->] (B) -- (A) node[midway, left] {$b_j$};
                \draw[->] (C) -- (A) node[midway, right] {$b_k$};
                
                % Natural transformation arrow
                \node at (1.5, 1.2) {$\cong$};
                % \node at (1.7, 1.2) {$\eta$};
            \end{tikzpicture}
        \end{center}

    There is a unit
    \[
        \eta \colon \ast \to \B(1)
    \]
    which is defined by the identity morphism $\id_X \in \Mul_{\B_{col}}^{b_1}(\{X\}, X)$.
    The unitality condition is satisfied as the identity morphism $\id_X$ is the left and right unit in $\Mul_{\B_{col}}^{b_1}(\{X\}, X)$.
    The associativity condition is also satisfied as it holds for $\B_{col}$: For any choice of integers
            \[
                n,  \ j_1, \dots, j_k, \ i_1, \dots, i_j
            \]
            with
            \[
                \sum_{1 \leq s \leq k} j_s = j, ~ \sum_{1 \leq r \leq j} i_r = i, \ g_s = \sum_{0 \leq l \leq s} j_l \ \text{and} \ h_s = \sum_{1 \leq q \leq j_s} i_{g_{s-1}+q}
            \]
            % \sum_{1 \leq s \leq k} j_s = j$, $\sum_{1 \leq r \leq j} i_r = i$, $g_s = \sum_{0 \leq l \leq s} j_l$ and $h_s = \sum_{1 \leq q \leq j_s} i_{g_{s-1}+q}$ 
            
            the following diagram
                \begin{center}
                \begin{tikzcd}
                    \B(k) \times (\prod_{s=1}^k \B(j_s)) \times (\prod_{r=1}^j \B(i_r)) \arrow[rr, "\gamma \times \id"] \arrow[dd, "s"']               &  & \B(j) \times (\prod_{r=1}^j \B(i_r)) \arrow[d, "\gamma"]   \\                                                                           &  & \B(i)   \\
                    \B(k) \times \prod_{s=1}^k (\B(j_s) \times \prod_{q=1}^{j_s} \B(i_{g_{s-1}+q})) \arrow[rr, "\id \times (\times_s \gamma)"'] &  & \B(k) \times (\prod_{s=1}^k \B(h_s)) \arrow[u, "\gamma"']
                \end{tikzcd}
                \end{center}
                where $s$ shuffles the entries of the tensor product, 
    commutes as the composition map $\gamma$ in $\B_{col}$ is associative for the following maps
    for $1 \leq s \leq k$, $1 \leq q \leq j_s$ and $h'_q = \sum_{t=1}^{g_{s-1}+1} i_t$:
    \begin{align*}
        \alpha \colon \langle i \rangle &\to \langle j \rangle \\
        h'_{q-1}+1, \dots, h'_q &\mapsto g_{s-1}+q \\
    \end{align*}
    with $\widetilde{\alpha_{g_{s-1}+q}} = b_{i_{g_{s-1}+q}}$ and
    \begin{align*}
        \alpha' \colon \langle j \rangle &\to \langle k \rangle \\
        g_{s-1}+1, \dots, g_s &\mapsto s \\
    \end{align*}
    with $\widetilde{\alpha'_s} = b_{j_s}$.
    So we have defined a planar operad $\B$. It remains to see how the braid action on $\B$ is induced by the
    definition of $\B_{col}$.

    Similarily to the symmetric case, in a colored braided operad (even with one color), the set of morphisms 
    depends not only on $n$, the cardinality of the indexing set, but also on the specific choice of finite 
    indexing set as well as the corresponding configuration. Thus, for a different labeling $I_1$ and
    $I_2$ with $|I_1|=n=|I_2|$ and different configurations $\widetilde{I_1}$ and $\widetilde{I_2}$ we 
    have potentially different sets of morphisms:
    \[
        \Mul_{\B_{col}}^{\widetilde{I_1}}(\{X\}_{i \in I_1}, X) \text{ and } \Mul_{\B_{col}}^{\widetilde{I_2}}(\{X\}_{i \in I_2}, X)
    \]
    
    This raises the question of how the structures of a one colored braided operad and 
    a braided operad reconcile. The reason these definitions are still equivalent 
    lies in the induced braid group action on the set of morphisms. \todo{satz umschreiben?}



    The braiding of the inputs ensures that any specific labeling of the inputs as represented 
    by $I_1$ with $\widetilde{I_1}$ and $I_2$ with $\widetilde{I_2}$ can be transformed into each other via the action 
    of some $\sigma \in B_n$ which is represented by the following commutative diagram
    \begin{center}
        \begin{tikzpicture}
            % Define node positions
            \node (B) at (0, 2) {$I$};
            \node (C) at (3, 2) {$J$};
            \node (A) at (1.5, 0) {$\langle 1 \rangle$};
            
            % Draw arrows
            \draw[->] (B) -- (C) node[midway, above] {$\bar{\sigma}$};
            \draw[->] (B) -- (A) node[midway, left] {$\widetilde{I}$};
            \draw[->] (C) -- (A) node[midway, right] {$\widetilde{J}$};
            
            % Natural transformation arrow
            \node[rotate=-40] at (1.4, 1.2) {$\Downarrow \sigma$};
            % \node at (1.7, 1.2) {$\eta$};
        \end{tikzpicture}
    \end{center}
    where $\bar{\sigma}$ is a relabeling and $\sigma$ is the path between the configurations 
    $\widetilde{I}$ and $\widetilde{J}$.
    Hence, although the morphisms in the colored braided operad are indexed by specific sets and configurations, they 
    behave the same way up to this braid action. Therefore, the group action effectively collapses the dependence 
    on the specific finite sets and their configurations.
