\section{Colored Braided Operads}

In this section, we will define colored braided operads and will aim to find a
functor that encodes all the necessary data to give an alternative definition so
the notion can eventually be lifted to an $\infty$-categorical context.

\subsection{Colored Braided Operads}

In this section we finally give our definition of a colored braided operad and see
a first example. Before we get started, we need to introduce some terminology.

\begin{defin}
    Let $I$ be a finite set. For a bijection of finite sets $\alpha \colon I
    \cong \langle n \rangle^{\circ}$, we call a configuration $c \in \E_2(n)$ an 
    $I$-labeled configuration.
\end{defin}

\begin{defin}
    Let $I$ be a finite set, $c$ and $c'$ $I$-labeled
    configurations corresponding to the same bijection $I \cong \langle n
    \rangle^{\circ}$. A path from $c$ to $c'$ is a path between the
    configurations in $\E_2(n)$, i.e. a continuous map
    \[
        f \colon [0,1] \to \E_2(n)
    \] with $f(0)=c$ and $f(1)=c'$.
\end{defin}


\begin{defin}
    Let $\Br$ be the following $\Grp$-enriched category:
    \begin{enumerate}[(i)]
        \item The objects of $\Br$ are finite sets.
        \item A morphism $\alpha \colon I \to J$ in $\Br$ consists of a map of
        finite sets  $\alpha' \colon I \to J$ and a collection of configurations

        $$\left\{\widetilde{\alpha}_j \in
        \E_2\left(|\alpha'^{-1}(j)|\right)\right\}_{j \in J}$$

        where each configuration $\widetilde{\alpha}_j$ is
        $\alpha'^{-1}(j)$-labeled.
        \begin{enumerate}
            \item For morphisms
        \[
            \alpha \colon I \to J, \quad \beta \colon J \to K
        \]
        
        the composition $\beta \circ \alpha$ consists of the map of finite sets
        \[
            \beta' \circ \alpha' \colon I \to K
        \]
        and for all $k \in K$ with
        \[
            \beta'^{-1}(k)=\{j_1, \dots, j_p\}
        \]
        the $(\beta' \circ \alpha')^{-1}(k)$-labeled configuration
        \[
            \widetilde{(\beta \circ \alpha)}_k \coloneqq \gamma(\widetilde{\beta}_k, \widetilde{\alpha}_{j_1}, \dots, \widetilde{\alpha}_{j_p}),
        \]
        where $\gamma$ is the composition map in $\E_2$.
        
        % analogously to the composition map in $\Ex$ as defined above.
        \item The identity morphism $\id_I$ is given by the identity on the
        finite set $I$ and the collection of trivial configurations $\{\id^{(i)}
        \in \E_2(1)\}_{i \in I}$.
        \end{enumerate}
        \item For morphisms $\alpha, \beta \colon I \to J$ in $\Br$ with
        $\alpha' = \beta'$ as maps of finite sets. A $2$-morphism in $\Br$
        consists of a family of homotopy classes of paths
        \[
            \left\{\eta_j \colon \widetilde{\alpha}_i \Rightarrow \widetilde{\beta}_j \right\}_{j \in J}.
        \]
    \end{enumerate}
\end{defin}
This defines a strict $2$-category.

\begin{rem}\label{rem:e2_1_paths} We note the following about configurations $c,
    c' \in \E_2(1)$. There is exactly one homotopy class of paths from $c$ to $c'$
    as the unit disk is contractible. This means that for diagrams in $\Br$ the
    following assumption can be made: Let $I$, $J$ be finite sets,
    $\widetilde{I}$ $I$-labeled $\widetilde{J}$ $J$-labeled configurations and

\end{rem}
    

We are now ready to define a colored braided operad:   

\begin{defin}
    A {\em colored braided operad} $\B$ consists of the following data:
    \begin{enumerate}[(i)]
        \item A collection $\{X,Y,Z, \dots\}$ of colors.
        \item For every finite set $I$, $I$-labeled configuration
        $\widetilde{I}$, every collection of colors $\{X_i\}_{i \in I}$, and a
        color $Y$ in $\B$, a set
        \[
            \Mul_{\B}^{\widetilde{I}}(\{X_i\}_{i \in I}, Y)
        \]
        called the set of morphisms from $\{X_i\}_{i \in I}$ to $Y$.
        \item For $\alpha \colon I \to J$ being a morphism in $\Br$ with its
        collection of configurations labeled by the preimages of $\alpha$: 
        $$\widetilde{\alpha} = (\widetilde{I_j})_{j \in J}$$ where $I_j$ denotes
        the fiber over $j$ for all $j \in J$, a $2$-morphism $\eta$:
        \begin{center}
            \begin{tikzpicture}
                % Define node positions
                \node (B) at (0, 2) {$I$}; \node (C) at (3, 2) {$J$}; \node (A)
                at (1.5, 0) {$\{ 1 \}$};
                
                % Draw arrows
                \draw[->] (B) -- (C) node[midway, above] {$\widetilde{\alpha}$};
                \draw[->] (B) -- (A) node[midway, left] {$\widetilde{I}$};
                \draw[->] (C) -- (A) node[midway, right] {$\widetilde{J}$};
                
                % Natural transformation arrow
                \node[rotate=-40] at (1.5, 1.2) {$\Downarrow \eta$};
                % \node at (1.7, 1.2) {$\eta$};
            \end{tikzpicture}
        \end{center}
        a collections of colors $\{X_i\}_{i \in I}$, $\{Y_j\}_{j \in J}$ and
        color $Z$, a composition map:
        \[
            \gamma \colon \prod_{j \in J} \Mul_{\B}^{\widetilde{I_j}}\left(\{X_i\}_{i \in I_j}, Y_j\right) \times \Mul_{\B}^{\widetilde{J}}\left(\{Y_j\}_{j\in J}, Z\right) \to \Mul_{\B}^{\widetilde{I}}\left(\{X_i\}_{i \in I}, Z\right)  
        \]
        \item For any $I$-labeled configuration c with $|I|=1$, a collection of
        morphisms
        $$\{ \id_X \in \Mul_{\B}^{c}(\{X\}, X)\}_{X \in \B}$$ called the
        identity morphisms.
        \end{enumerate}
        such that the following properties hold:
        \begin{enumerate}
            \item The identity morphisms are a left and right unit with respect
        to the composition map $\gamma$, i.e. the following diagrams commute:
        \begin{center}
            \begin{tikzcd}
                {\Mul_{\B}^{\widetilde{I}}(\{X_i\}_{i \in I}, Y)} \arrow[r,
                "\cong"] \arrow[d, "\id"] &
                {\Mul_{\B}^{\widetilde{I}}(\{X_i\}_{i \in I}, Y) \times
                \{\id_Y\}} \arrow[d, hook]           \\
                {\Mul_{\B}^{\widetilde{I}}(\{X_i\}_{i \in I}, Y)} &
                {\Mul_{\B}^{\widetilde{I}}(\{X_i\}_{i \in I}, Y) \times
                \Mul_{\B}^{c}((Y), Y)} \arrow[l, "\gamma"]
            \end{tikzcd}
        \end{center}
       
        \begin{center}
            \begin{tikzcd}
                {\Mul_{\B}^{\widetilde{I}}(X_I, Y)} \arrow[r, "\cong"] \arrow[d,
                "\id"] & {\left(\prod_{i \in I} \{\id_{X_i}\}\right) \times
                \Mul_{\B}^{\widetilde{I}}(X_I, Y)} \arrow[d, hook]           \\
                {\Mul_{\B}^{\widetilde{I}}(X_I, Y)} & {\left(\prod_{i \in I}
                \Mul_{\B}^{c}((X_i), X_i)\right) \times
                \Mul_{\B}^{\widetilde{I}}(X_I, Y)} \arrow[l, "\gamma"]
            \end{tikzcd}                                             
        \end{center}

        
        
    
        \item For every sequence of morphisms $I \overset{\alpha}{\to} J
        \overset{\beta}{\to} K$ in $\Br$, $I$-labeled configuration
        $\widetilde{I}$, $J$-labeled configuration $\widetilde{J}$, 2-morphisms
        $\eta$, $\epsilon$ in $\Br$ as in the following diagram:

        \begin{center}
            \begin{tikzpicture}
                % Define node positions
                \node (B) at (0, 2) {$I$}; \node (C) at (3, 2) {$J$}; \node (A)
                at (3, -1) {$\{1 \}$}; \node (D) at (6, 2) {$K$};

                % Draw arrows
                \draw[->] (B) -- (C) node[midway, above] {$\alpha$}; 
                \draw[->] (C) -- (D) node[midway, above] {$\beta$}; 
                \draw[->] (B) -- (A) node[midway, left] {$\widetilde{I}$}; 
                \draw[->] (C) -- (A)
                node[midway, right] {$\widetilde{J}$}; 
                \draw[->] (D) -- (A)
                node[midway, right] {$\widetilde{K}$};
                % Natural transformation arrow
                \node at (2.0, 1.1) {$\overset{\Leftarrow}{\eta}$}; \node at
                (4.2, 1.1) {$\overset{\Leftarrow}{\epsilon}$};
            \end{tikzpicture}
        \end{center}
        % \begin{center} \begin{tikzcd} {\prod_{j \in J} \Mul_{\O}(X_I, Y_j)
        %     \times \prod_{k \in K} \Mul_{\O}(Y_J, W_k) \times \Mul_{\O}(W_K,
        %     Z)} \arrow[rd, "\id \times \gamma"] \arrow[dd, "(\gamma \circ s)
        %     \times \id"] & \\
        %                     & {\prod_{j \in J} \Mul_{\O}(X_{I_j}, Y_j) \times
        %                     \Mul_{\O}(Y_J, Z)} \arrow[dd] \\
        %         {\prod_{k \in K} \Mul_{\O}(X_{I_k}, W_k) \times \Mul_{\O}(W_K,
        %         Z)} \arrow[rd] & \\
        %                     & {\Mul_{\O}(X_I, Z)}         
        %     \end{tikzcd} \end{center}

        collections of colors $X_I = \{X_i\}_{i \in I}$, $Y_J = \{Y_j\}_{j \in
        J}$, $W_K = \{W_k\}_{k \in K}$ and a color $Z \in \B$ the following
        diagram commutes:
        \begin{center}
        \begin{adjustbox}{max width=\textwidth}
            \begin{tikzcd}
                {\prod_{j \in J} \Mul_{\B}^{\widetilde{I_j}}(X_{I_j}, Y_j)
                \times \prod_{k \in K} \Mul_{\B}^{\widetilde{J_k}}(Y_{J_k}, W_k)
                \times \Mul_{\B}^{\widetilde{K}}(W_K, Z)} \arrow[rd, "\id \times
                \gamma"] \arrow[dd, "(\gamma \circ s) \times \id"] & \\
                            & {\prod_{j \in J}
                            \Mul_{\B}^{\widetilde{I_j}}(X_{I_j}, Y_j) \times
                            \Mul_{\B}^{\widetilde{J}}(Y_J, Z)} \arrow[dd] \\
                {\prod_{k \in K} \Mul_{\B}^{\widetilde{I_k}}(X_{I_k}, W_k)
                \times \Mul_{\B}^{\widetilde{K}}(W_K, Z)} \arrow[rd] & \\
                            & {\Mul_{\B}(X_I, Z)}         
            \end{tikzcd}
            \end{adjustbox}
        \end{center}
        \end{enumerate}

\end{defin}

\begin{eg}
    Let $\Bc$ be a colored braided operad with one color $X$. Similarly to the
    symmetric case, $\Bc$ induces a braided operad $\B$ as in Definition
    \ref{def-br-operad}. For $n \in \N$, we set
    \[
        \B(n) = \Mul_{\Bc}^{b_n}(\{X\}_{1 \leq i \leq n}, X)
    \]
    where $b_n$ is our basepoint configuration in $\E_2(n)$ as chosen in
    Construction \ref{con:unicover}. Further, we define the composition map
    \[
        \theta \colon \B(k) \times \B(j_1) \times \cdots \times \B(j_k) \to \B(j)
    \]
    where $j = j_1 + \cdots + j_k$, from the composition map $\gamma$ in $\Bc$:
    \[
        \gamma \colon  \prod_{1 \leq i \leq k} \Mul_{\Bc}^{b_{j_i}}\left(\{X\}^{j_i}, X\right) \times \Mul_{\Bc}^{b_k}\left(\{X\}^k, X\right) \to \Mul_{\Bc}^{b_j}\left(\{X\}^j, X\right)
    \]
    corresponding to the unique $\E_1$-translation from $b_j$ to $b_k(b_{j_1},
    \dots, b_{j_k})$ as defined in Remark \ref{rem:e1-translation}:
    \begin{center}
            \begin{tikzpicture}
                % Define node positions
                \node (B) at (0, 2) {$\langle j \rangle^{\circ}$}; \node (C) at (3, 2)
                {$\langle k \rangle^{\circ}$}; \node (A) at (1.5, 0) {$\{ 1 \}$};
                
                % Draw arrows
                \draw[->] (B) -- (C) node[midway, above] {$(b_{j_1}, \dots,
                b_{j_k})$}; \draw[->] (B) -- (A) node[midway, left] {$b_j$};
                \draw[->] (C) -- (A) node[midway, right] {$b_k$};
                
                % Natural transformation arrow
                \node at (1.5, 1.2) {$\cong$};

            \end{tikzpicture}
        \end{center}

    \[
        \eta \colon \ast \to \B(1)
    \]
    which is defined by the identity morphism $\id_X \in \Mul_{\Bc}^{b_1}(\{X\},
    X)$. The unitality condition is satisfied as the identity morphism $\id_X$
    is the left and right unit for any composition map $\gamma$, so in
    particular $\theta$. The associativity condition is also satisfied as it
    holds for $\Bc$:

Next, we extend $\Br$ with a distinguished point to the following category.
\begin{defin}
    Let $\Brx$ be the following $\Grp$-enriched category:
    \begin{enumerate}[(i)]
        \item The objects of $\Brx$ are finite sets $\langle n \rangle$ for $n
        \in \N$.
        \item A morphism $\alpha \colon \langle n \rangle \to \langle m \rangle$
        in $\Brx$ consists of a morphism $\alpha \colon \langle n \rangle \to
        \langle m \rangle$ in $\Fin$ and a collection of configurations

        $$\{\widetilde{\alpha}_j \in \E_2(|\alpha^{-1}(j)|)\}_{1 \leq j \leq
        m}$$

        where each configuration $\widetilde{\alpha}_j$ is labeled by
        $\alpha^{-1}(j)$. 
        \begin{enumerate}
            \item The composition of morphisms is defined via the composition map of $\E_2$. Analogously to the composition map in $\Ex$ from Remark
            \ref{rem:ex}.
            \item The identity morphism $\id_{\langle n \rangle}$ is the
            collection of trivial configurations $\{\id^{(j)} \in \E_2(1)\}_{1
            \leq j \leq n}$. 
        \end{enumerate}
        \item The morphisms in $\Brx$ form the groupoid
        \[
        \Hom_{\Brx}(\langle n \rangle, \langle m \rangle) = \coprod_{\substack{\alpha \colon \langle n \rangle \to \langle m \rangle \in \Fin}} \Hom_{\Brx}^{\alpha}(\langle n \rangle, \langle m \rangle)
        \]
        where 
        $$\Hom_{\Brx}^{\alpha}(\langle n \rangle, \langle m \rangle)$$ denotes
        the product of the groupoids of homotopy classes of paths in the
        configuration space $\E_2(|\alpha^{-1}(j)|)$ labeled by $\alpha^{-1}(j)$
        for all $1 \leq j \leq m$.
    \end{enumerate}
\end{defin}
This defines a strict $2$-category.

\begin{defin}
    A morphism $\alpha \colon \langle n \rangle \to \langle m \rangle$ in $\Brx$
    is called active if it is active in $\Fin$, i.e. $\alpha^{-1}(\ast) =
    \{\ast\}$.
\end{defin}

\begin{rem}
    We can extend the definition of $\Brx$ to include any finite set $I$ with a
    distinguished point. However, note that that category would be equivalent to
    $\Brx$ in the analogous way as $\Fin$ is equivalent to the category of
    finite sets with a distinguished point.
\end{rem}

\begin{defin}
    Let $\B$ be a colored operad. We define a category $\Bx$ in the following
    way:
    \begin{enumerate}[(i)]
        \item The objects of $\Bx$ are finite sequences of colors of $\B$.
        \item For $\{X_i\}_{1 \leq i \leq m}$, $\{Y_j\}_{1 \leq j \leq n}$ being
        objects in $\Bx$, a morphism 
        \[
            \phi \colon \{X_i\}_{1 \leq i \leq m} \to \{Y_j\}_{1 \leq j \leq n}
        \]
        is given by a map $\alpha_{\phi} \colon \langle m \rangle \to \langle n
        \rangle$ in $\Brx$ together with a family of morphisms
        \[
            \{ \phi_j \in \Mul_{\B}^{\widetilde{\alpha_j}}(\{X_i\}_{i \in \alpha^{-1}(j)}, Y_j)\}_{1 \leq j \leq n}
        \]
        in $\B$. We call $\alpha_{\phi}$ the underlying morphism of $\phi$.
        \item Composition of morphisms is determined by the composition in
        $\Brx$ and $\B$, i.e. for morphisms $\phi \colon X \to Y$, $\phi'
        \colon Y \to Z$ in $\Bx$  with $\alpha \colon \langle n \rangle \to
        \langle m \rangle$, $\alpha' \colon \langle m \rangle \to \langle k
        \rangle$ respectively, we define $\alpha'' = \alpha'\circ \alpha$
        \[
            \{(\phi' \circ \phi)_l \in  \Mul_{\B}^{\widetilde{\alpha''_l}}(\{X_i\}_{i \in \alpha''^{-1}(l)}, Z_l)\}_{1 \leq l \leq k}
        \]
        where $(\phi' \circ \phi)_l$ is applying to $((\phi_i)_{i \in
        \alpha''^{-1}(l)}, \phi'_l)$ the following composition map 
        \[
            \gamma \colon \prod_{j \in \alpha'^{-1}(l)} \Mul_{\B}^{\widetilde{\alpha_j}}(\{X_i\}_{i \in \alpha^{-1}(j)},Y_j) \times \Mul_{\B}^{\widetilde{\alpha'_l}}(\{Y_i\}_{i \in \alpha'^{-1}(l)},Z_l) \to \Mul_{\B}^{\widetilde{\alpha''}_l}(\{X_i\}_{i \in \alpha''^{-1}(l)},Z_l).
        \] 
    \end{enumerate}
\end{defin}

Let $\B$ be a colored braided operad, then there is a forgetful functor
\[
    \pi \colon \Bx \to \Brx,
\]
defined by:
\begin{align*}
    [X_1, \dots, X_n] &\mapsto \langle n \rangle \\
    (f \colon [X_1, \dots, X_n] \to [X_1', \dots, X_m']) &\mapsto (\alpha_f \colon \langle n \rangle \to \langle m \rangle)
\end{align*}
where $\alpha_f$ refers to the morphism in $\Brx$ underlying $f$. It is straight
forward to see that this defines a functor.
